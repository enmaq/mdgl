cmake_minimum_required(VERSION 3.16)
project(mdgl VERSION 0.3.0 LANGUAGES C)
add_library(mdgl mdgl.c mdgl_render.c mdgl_play.c)
target_compile_definitions(mdgl PRIVATE _GNU_SOURCE)
target_link_libraries(mdgl PUBLIC m)
target_include_directories(mdgl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(mdgl PRIVATE -Wall -Wextra)
enable_testing()
add_executable(mdgl_check_test test_check.c)
target_link_libraries(mdgl_check_test PRIVATE mdgl m)
add_test(NAME check COMMAND mdgl_check_test)
add_executable(mdgl_render_test test_render.c)
target_link_libraries(mdgl_render_test PRIVATE mdgl m)
add_test(NAME render COMMAND mdgl_render_test)
# Locale discriminator: skipped (77) when the comma-decimal locale is not installed.
add_executable(mdgl_locale_test test_locale.c)
target_link_libraries(mdgl_locale_test PRIVATE mdgl m)
add_test(NAME locale COMMAND mdgl_locale_test de_DE.UTF-8)
set_tests_properties(locale PROPERTIES SKIP_RETURN_CODE 77)
# Allocation-failure injection needs GNU ld --wrap.
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND NOT APPLE)
  add_executable(mdgl_oom_test test_oom.c)
  target_link_libraries(mdgl_oom_test PRIVATE mdgl m -Wl,--wrap=malloc -Wl,--wrap=calloc)
  add_test(NAME oom COMMAND mdgl_oom_test)
endif()
# Conformance adapter (NDJSON, protocol v1) and its transport tests.
add_executable(mdgl_adapter mdgl_adapter.c)
target_link_libraries(mdgl_adapter PRIVATE mdgl m)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME adapter COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_adapter.py $<TARGET_FILE:mdgl_adapter>)
endif()
# Example host: cmake --build <dir> --target mdgl_render; run from c/: <dir>/mdgl_render
add_executable(mdgl_render examples/render.c)
target_link_libraries(mdgl_render PRIVATE mdgl m)
add_test(NAME example COMMAND mdgl_render ${CMAKE_CURRENT_SOURCE_DIR}/../examples/text.mdgl)

# Command line: installed as `mdgl` (mdgl check|frames|play FILE). The library target is also named mdgl,
# the executable only differs by file name (mdgl vs libmdgl.a).
add_executable(mdgl_cli mdgl_cli.c)
target_compile_definitions(mdgl_cli PRIVATE _GNU_SOURCE)
set_target_properties(mdgl_cli PROPERTIES OUTPUT_NAME mdgl)
target_link_libraries(mdgl_cli PRIVATE mdgl m)
include(GNUInstallDirs)
install(TARGETS mdgl_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS mdgl ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES mdgl.h mdgl_play.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md THIRD_PARTY_NOTICES.md LICENSE-MIT LICENSE-APACHE DESTINATION ${CMAKE_INSTALL_DOCDIR})
add_executable(mdgl_play_test test_play.c)
target_link_libraries(mdgl_play_test PRIVATE mdgl m)
add_test(NAME play COMMAND mdgl_play_test)
if(Python3_FOUND AND UNIX)
  add_test(NAME play_pty COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_play_pty.py $<TARGET_FILE:mdgl_cli>)
endif()
